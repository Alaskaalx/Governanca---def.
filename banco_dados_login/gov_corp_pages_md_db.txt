-- ========================================================
-- 1. CRIAÇÃO DO BANCO E LIMPEZA (CUIDADO: Apaga dados anteriores)
-- ========================================================
CREATE DATABASE IF NOT EXISTS gov_corp_pages_gm_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE gov_corp_pages_gm_db;

DROP TABLE IF EXISTS medidas_disciplinares_consolidado;
DROP TABLE IF EXISTS atos_faltosos;
DROP TABLE IF EXISTS grupos;

-- ========================================================
-- 2. TABELAS DE REGRAS E DOMÍNIOS (Grupos e Atos)
-- ========================================================

-- Tabela de Grupos
CREATE TABLE grupos (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    nome_grupo VARCHAR(100) NOT NULL
);

-- Tabela de Atos Faltosos (Com todas as regras do Compliance)
CREATE TABLE atos_faltosos (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    grupo_id INTEGER,
    descricao_ato VARCHAR(255) NOT NULL,
    item VARCHAR(50) NOT NULL,
    gravidade VARCHAR(50) NOT NULL,
    medida_1a_ocorrencia VARCHAR(100) NOT NULL,
    descricao_observacao TEXT NOT NULL,
    FOREIGN KEY (grupo_id) REFERENCES grupos(id)
);

-- ========================================================
-- 3. TABELA SUPER CONSOLIDADA (Histórico + Base de Colaboradores)
-- ========================================================
CREATE TABLE medidas_disciplinares_consolidado (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    
    -- DADOS DO COLABORADOR (Usados no Autopreenchimento e Dashboard)
    matricula INTEGER NOT NULL,
    cpf VARCHAR(14),
    concre VARCHAR(100),
    id2 VARCHAR(100),
    filial VARCHAR(100),
    nome VARCHAR(255),
    cargo VARCHAR(150),
    data_admissao DATE,
    registro_func VARCHAR(50),
    situacao_funcionario VARCHAR(50),
    centro_custo VARCHAR(100),

    -- DADOS DA MEDIDA / OCORRÊNCIA
    proxima_acao VARCHAR(100),
    data_inicio_suspensao DATE,
    data_fim_suspensao DATE,
    data_inicio_ocorrencia DATE,
    data_fim_ocorrencia DATE,
    validade_medida VARCHAR(100),
    nro_chamado_sagoglobal VARCHAR(100),
    data_ato_faltoso DATE,
    
    -- DADOS DA INFRAÇÃO
    grupo_id INTEGER,
    ato_faltoso_id INTEGER,
    classificacao_ato_faltoso VARCHAR(255),
    desc_ato_faltoso TEXT,
    tipo VARCHAR(100),
    gravidade VARCHAR(100),
    quantidade_faltas INTEGER DEFAULT 0,
    observacao TEXT,

    -- HIERARQUIA
    gestor_1 VARCHAR(255),
    gestor_2 VARCHAR(255),
    gerente VARCHAR(255),
    diretor VARCHAR(255),

    -- TEXTOS E CONTROLE
    info_compliance TEXT,
    texto_gerado TEXT,
    data_registro DATETIME DEFAULT CURRENT_TIMESTAMP,
    
    -- LIGAÇÕES COM AS REGRAS OFICIAIS
    FOREIGN KEY (grupo_id) REFERENCES grupos(id),
    FOREIGN KEY (ato_faltoso_id) REFERENCES atos_faltosos(id)
);
